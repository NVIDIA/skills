// One thread per vocab entry; occurrences of token v in the history are stored CSR-style.
__global__ void sampling_penalty_kernel(float* logits, const int* occ_offsets, const int* occ_pos,
                                        const float* decay, int vocab, int batch) {
  int b = blockIdx.y;
  int v = blockIdx.x * blockDim.x + threadIdx.x;
  if (v >= vocab) return;
  const int* off = occ_offsets + (size_t)b * (vocab + 1);
  float acc = 0.f;
  for (int k = off[v]; k < off[v + 1]; ++k) {        // trip count = #occurrences of token v
    int p = occ_pos[k];
    acc += (p < 64) ? decay[p] : decay[63] * 0.5f;   // data-dependent branch
  }
  if (acc > 0.f) logits[(size_t)b * vocab + v] -= acc;
}
