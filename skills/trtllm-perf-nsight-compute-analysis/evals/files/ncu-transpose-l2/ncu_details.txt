// out is [N, M] row-major, x is [M, N] row-major; out[i][j] = gelu(x[j][i])
__global__ void transpose_gelu_kernel(const float* __restrict__ x,
                                      float* __restrict__ out, int M, int N) {
  int j  = blockIdx.x * 32 + threadIdx.x;   // output column (contiguous in out)
  int i0 = blockIdx.y * 32 + threadIdx.y;   // first output row handled by this thread
  for (int k = 0; k < 32; k += 8) {
    int i = i0 + k;
    if (i < N && j < M) {
      float v = x[(size_t)j * N + i];        // lanes of a warp differ in j
      out[(size_t)i * M + j] = 0.5f * v * (1.0f + erff(v * 0.70710678f));
    }
  }
}
